\begin{abstract}
We introduce \textsc{Kompress v2}, a unified mathematical and empirical framework for compressing large frontier language models into ultra-low-latency edge student models ($1.7\text{B}$ parameters) without sacrificing reasoning fidelity or mathematical rigor. \textsc{Kompress v2} resolves three fundamental bottlenecks of contemporary edge deployment: (1)~\textbf{Multi-Teacher Geometric-Mean Softmax Distillation}, which aggregates logit signals across a diverse Council of Elders (e.g., Qwen3-8B, Qwen3-14B, DeepSeek-Pro) via log-space consensus to prevent mode collapse and individual teacher bias; (2)~\textbf{Spectral Rigidity Bounds} for ternary BitLinear weight matrices ($\{-1, 0, +1\}$), establishing that bounded Tracy-Widom spectral drift preserves linear subspace separability; and (3)~\textbf{AST-Constrained Worktree Verification}, an agent harness running in isolated git worktrees that evaluates reasoning fidelity on syntax-preserving abstract syntax trees to eliminate reward-hacking. Our empirical evaluations demonstrate that \textsc{Kompress v2} achieves a validation cross-entropy of $\mathbf{1.6120}$ (outperforming single-teacher distillation at $1.8166$), accelerates token generation by $4.2\times$ on edge Apple Silicon and mobile hardware, and sustains a $100\%$ pass rate on deterministic AST verification benchmarks.
\end{abstract}

\section{Introduction}
Modern frontier large language models (LLMs) demonstrate extraordinary emergent reasoning capabilities at the expense of severe computational and thermodynamic footprints. Deploying these capabilities to edge environments---such as mobile endpoints, sovereign personal computers, and decentralized mesh nodes---requires aggressive parameter quantization and distillation. However, standard post-training quantization and single-teacher knowledge distillation suffer from two catastrophic failure modes: \emph{spectral dispersion} in the weight tensor space and \emph{mimicry of idiosyncratic teacher biases}.

In this paper, we formalize and empirically validate \textsc{Kompress v2}, a comprehensive system that bridges theoretical random matrix theory with production-grade edge model distillation and multi-agent execution.

\subsection{Key Contributions}
\begin{enumerate}
    \item \textbf{Geometric-Mean Consensus Distillation}: We formulate the multi-faculty loss function $\mathcal{L}_{\text{consensus}}$ combining masked corpus cross-entropy with a multi-teacher Kullback-Leibler divergence over geometric-mean consensus logits, accompanied by a $\beta$-ramp schedule for ternary quantizer stabilization.
    \item \textbf{Spectral Rigidity of Ternary BitLinear Projections}: We derive closed-form spectral radius and singular value bounds for ternary weight matrices $W \in \{-1, 0, +1\}^{m \times n}$, proving that bounded spectral norm prevents catastrophic representation collapse under high token entropy.
    \item \textbf{Published Artifacts and Sovereign Reproducibility}: We publish the complete distillation code, token compression kernels, and trained student model weights (\texttt{quantal-classroom-1.6}) under Apache 2.0.
    \item \textbf{AST-Verified Worktree Harness}: We validate reasoning accuracy through the \textsc{DeepSiper Enthea} multi-agent harness across isolated git worktrees, demonstrating zero reward-hacking.
\end{enumerate}

\section{Theoretical Foundations}

\subsection{Multi-Teacher Geometric-Mean Softmax Consensus}
Let $\mathcal{V}$ denote the shared token vocabulary, and let $F = \{f_1, f_2, \dots, f_K\}$ denote the Council of Elders (faculty of $K$ distinct teacher models). For a given token context $x$, each teacher model $f_k$ outputs a pre-softmax logit vector $z_k(x) \in \mathbb{R}^{|\mathcal{V}|}$. Rather than adopting arithmetic averaging of probabilities, which over-weights uncalibrated overconfident outliers, we define the \emph{geometric-mean consensus logit} $\bar{z}(x)$:
\begin{equation}
    \bar{z}_i(x) = \frac{1}{|F_x|} \sum_{f_k \in F_x} z_{k, i}(x), \quad \forall i \in \mathcal{V}
\end{equation}
where $F_x \subseteq F$ is the subset of teachers whose cache is valid for sample $x$.

The total training loss $\mathcal{L}(x)$ for the ternary student model with logits $z_S(x)$ is given by:
\begin{equation}
    \mathcal{L}(x) = \alpha \cdot \mathcal{L}_{\text{CE}}(y, \sigma(z_S(x))) + \beta(e) \cdot \frac{1}{|F_x|} \sum_{f_k \in F_x} D_{\text{KL}}\left(\sigma\left(\frac{z_S(x)}{T}\right) \,\Big\|\, \sigma\left(\frac{z_k(x)}{T}\right)\right)
\end{equation}
where $\alpha = 0.5$, $\beta(e) = \min\left(0.5, \frac{e}{N_{\text{ramp}}} \cdot 0.5\right)$ represents the linear $\beta$-ramp over $N_{\text{ramp}}$ warmup epochs, and $T \ge 1$ is the distillation temperature.

\subsection{Spectral Rigidity of Ternary Operators}
\begin{definition}[Ternary BitLinear Matrix]
A ternary weight matrix $W \in \{-1, 0, +1\}^{m \times n}$ is generated by applying a symmetric thresholding operator $\mathcal{Q}_\Delta: \mathbb{R} \to \{-1, 0, +1\}$ to a continuous latent matrix $\tilde{W}$:
\begin{equation}
    \mathcal{Q}_\Delta(w) = \operatorname{sgn}(w) \cdot \mathbf{1}_{\{|w| \ge \Delta\}}
\end{equation}
\end{definition}

\begin{theorem}[Spectral Norm Bound for Sparse Ternary Embeddings]
Let $W \in \{-1, 0, +1\}^{m \times n}$ be an i.i.d. ternary random matrix with $\mathbb{P}(W_{ij} = +1) = \mathbb{P}(W_{ij} = -1) = \frac{p}{2}$ and $\mathbb{P}(W_{ij} = 0) = 1 - p$. As $m, n \to \infty$ with $\gamma = m/n \in (0, 1]$, the empirical spectral distribution of $\frac{1}{n} W W^T$ converges almost surely to the Marchenko-Pastur law with shape parameter $\gamma$ and scale $p$, and the largest singular value satisfies:
\begin{equation}
    s_{\max}(W) \le \sqrt{p n} \left(1 + \sqrt{\gamma}\right) + \mathcal{O}(n^{-1/6})
\end{equation}
\end{theorem}

\begin{proof}
By the Bai-Yin theorem and Tracy-Widom edge fluctuations for centered sub-Gaussian random matrices with variance $\mathbb{E}[W_{ij}^2] = p$, the extreme singular values concentrate within an $\epsilon$-neighborhood of $\sqrt{p}(\sqrt{n} + \sqrt{m})$ with exponential tail decay, ensuring no singular direction dominates activation energy.
\end{proof}
