\begin{abstract}
We introduce \textsc{Kompress v2}, a mathematically grounded framework for distilling and compressing frontier language models into ultra-low-latency $1.7\text{B}$ edge student models with ternary weights ($\{-1, 0, +1\}$). \textsc{Kompress v2} provides three central contributions: (1)~\textbf{Consensus Equivalence in Multi-Teacher Distillation}: we prove that arithmetic averaging of teacher logit vectors induces an exact normalized geometric-mean probability distribution, and that minimizing the mean reverse Kullback-Leibler divergence to individual teachers is optimization-equivalent to minimizing reverse KL divergence against this geometric-mean consensus; (2)~\textbf{Spectral Rigidity \& Restricted Isometry Null Models}: we establish asymptotic Marchenko-Pastur bulk and extreme singular value limits for sparse ternary projection ensembles, identifying the bounded spectral window $0 < c \le s_i(W|_{\mathcal{S}}) \le C < \infty$ required on active semantic subspaces to prevent representation explosion and dimensional collapse; and (3)~\textbf{Structural Invariant Verification}: we formulate a four-predicate admissibility contract $\mathcal{A}(c) = \mathcal{P}(c) \land \mathcal{S}(c) \land \mathcal{T}(c) \land \mathcal{G}(c)$ across isolated git worktrees, decomposing reasoning fidelity into predictive, structural, and behavioral components. Empirically, \textsc{Kompress v2} achieves a validation cross-entropy of $\mathbf{1.6120}$ on edge reasoning benchmarks (improving from $2.1369$ base and $1.8166$ single-teacher KD), reduces resident memory from $3.4\text{ GB}$ to $\mathbf{0.42\text{ GB}}$, accelerates Apple Silicon M-series inference from $28$ to $\mathbf{118\text{ tokens/s}}$, and yields a $100\%$ verification pass rate across deterministic algorithmic test suites.
\end{abstract}

\section{Introduction}
Compressing high-parameter frontier reasoning models into edge-deployable form factors requires balancing three distinct dimensions of fidelity: \emph{predictive fidelity} $\mathcal{F}_{\text{pred}}$ (distributional alignment with supervisory models), \emph{structural fidelity} $\mathcal{F}_{\text{struct}}$ (syntactic and invariant preservation of generated programs), and \emph{behavioral fidelity} $\mathcal{F}_{\text{behav}}$ (task completion on target domains).

Standard distillation techniques typically supervise a student model with a single teacher under KL divergence or train ternary networks without formal spectral bounds. In this paper, we formalize the mathematical mechanisms underlying multi-faculty distillation, analyze ternary operators through random matrix null models, and verify edge execution via isolated worktree harnesses.

\section{Consensus Distillation Theory}

\subsection{Logit Averaging Induces Geometric-Mean Probability Consensus}
Let $\mathcal{V}$ denote the shared token vocabulary ($|\mathcal{V}| = V$), and let $F = \{f_1, f_2, \dots, f_K\}$ denote the Council of Elders (a faculty of $K$ distinct teacher models). For a given context $x$, each teacher outputs a logit vector $z_k(x) \in \mathbb{R}^V$.

\begin{proposition}[Consensus Equivalence]
Let teacher probability distributions at temperature $T > 0$ be defined by $p_k(i) = \frac{\exp(z_{k,i} / T)}{\sum_{j=1}^V \exp(z_{k,j} / T)}$. The softmax over the arithmetic mean logit vector $\bar{z} = \frac{1}{K}\sum_{k=1}^K z_k$ satisfies:
\begin{equation}
    \operatorname{softmax}\left(\frac{\bar{z}}{T}\right)_i = \frac{\left(\prod_{k=1}^K p_k(i)\right)^{1/K}}{\sum_{j=1}^V \left(\prod_{k=1}^K p_k(j)\right)^{1/K}} = \bar{p}(i)
\end{equation}
That is, logit averaging induces the normalized geometric-mean probability distribution of the faculty.
\end{proposition}

\begin{proof}
By definition:
\begin{align*}
    \operatorname{softmax}\left(\frac{\bar{z}}{T}\right)_i &= \frac{\exp\left(\frac{1}{K}\sum_{k=1}^K \frac{z_{k,i}}{T}\right)}{\sum_{j=1}^V \exp\left(\frac{1}{K}\sum_{k=1}^K \frac{z_{k,j}}{T}\right)} = \frac{\prod_{k=1}^K \exp(z_{k,i}/T)^{1/K}}{\sum_{j=1}^V \prod_{k=1}^K \exp(z_{k,j}/T)^{1/K}}
\end{align*}
Substituting $\exp(z_{k,i}/T) = p_k(i) \cdot \left(\sum_{m=1}^V \exp(z_{k,m}/T)\right) = p_k(i) \cdot Z_k$:
\begin{align*}
    \operatorname{softmax}\left(\frac{\bar{z}}{T}\right)_i &= \frac{\prod_{k=1}^K [p_k(i) Z_k]^{1/K}}{\sum_{j=1}^V \prod_{k=1}^K [p_k(j) Z_k]^{1/K}} = \frac{\left(\prod_{k=1}^K Z_k^{1/K}\right) \prod_{k=1}^K p_k(i)^{1/K}}{\left(\prod_{k=1}^K Z_k^{1/K}\right) \sum_{j=1}^V \prod_{k=1}^K p_k(j)^{1/K}} = \bar{p}(i)
\end{align*}
as the scalar partition pre-factor $\prod_{k=1}^K Z_k^{1/K}$ cancels identically.
\end{proof}

\begin{proposition}[Optimization Equivalence of Mean Reverse KL]
Let $q$ denote the student probability distribution. Minimizing the average reverse KL divergence across individual teachers is optimization-equivalent to minimizing reverse KL divergence against the normalized geometric-mean consensus $\bar{p}$:
\begin{equation}
    \frac{1}{K}\sum_{k=1}^K D_{\text{KL}}(q \,\|\, p_k) = D_{\text{KL}}(q \,\|\, \bar{p}) - \log Z
\end{equation}
where $Z = \sum_{j=1}^V \left(\prod_{k=1}^K p_k(j)\right)^{1/K}$ is a student-independent normalization constant.
\end{proposition}

\begin{proof}
Expanding the left-hand side:
\begin{align*}
    \frac{1}{K}\sum_{k=1}^K D_{\text{KL}}(q \,\|\, p_k) &= \frac{1}{K}\sum_{k=1}^K \sum_{i=1}^V q(i) \log \frac{q(i)}{p_k(i)} = \sum_{i=1}^V q(i) \log q(i) - \sum_{i=1}^V q(i) \left[ \frac{1}{K}\sum_{k=1}^K \log p_k(i) \right] \\
    &= \sum_{i=1}^V q(i) \log q(i) - \sum_{i=1}^V q(i) \log \left( \prod_{k=1}^K p_k(i)^{1/K} \right)
\end{align*}
Since $\bar{p}(i) = \frac{1}{Z}\prod_{k=1}^K p_k(i)^{1/K}$, we have $\prod_{k=1}^K p_k(i)^{1/K} = Z \cdot \bar{p}(i)$. Substituting:
\begin{align*}
    \frac{1}{K}\sum_{k=1}^K D_{\text{KL}}(q \,\|\, p_k) &= \sum_{i=1}^V q(i) \log \frac{q(i)}{\bar{p}(i)} - \sum_{i=1}^V q(i) \log Z = D_{\text{KL}}(q \,\|\, \bar{p}) - \log Z
\end{align*}
Because $\nabla_q \log Z = 0$, we obtain $\arg\min_q \frac{1}{K}\sum_{k=1}^K D_{\text{KL}}(q \,\|\, p_k) = \arg\min_q D_{\text{KL}}(q \,\|\, \bar{p})$.
\end{proof}

\begin{corollary}[Consensus Mode Suppression]
Because reverse KL divergence $D_{\text{KL}}(q \,\|\, \bar{p})$ penalizes assigning probability mass where the supervisory target $\bar{p}(i) \approx 0$, and $\bar{p}(i) \propto \prod_{k=1}^K p_k(i)^{1/K}$, any token candidate $i$ assigned near-zero probability by even a single teacher ($p_m(i) \to 0$) is strongly suppressed in the consensus target ($\bar{p}(i) \to 0$). This mechanism enforces conservative agreement and suppresses teacher-specific idiosyncratic modes.
\end{corollary}

\section{Spectral Rigidity \& Random Matrix Null Models}

\subsection{Marchenko-Pastur Asymptotics for Ternary Ensembles}
We consider a ternary random matrix null ensemble $W \in \{-1, 0, +1\}^{m \times n}$ with independent entries:
\begin{equation}
    \mathbb{P}(W_{ij} = +1) = \mathbb{P}(W_{ij} = -1) = \frac{p}{2}, \quad \mathbb{P}(W_{ij} = 0) = 1 - p
\end{equation}
so that $\mathbb{E}[W_{ij}] = 0$ and $\operatorname{Var}(W_{ij}) = p$.

\begin{theorem}[Spectral Edges for Sparse Ternary Matrices]
Let $m, n \to \infty$ with aspect ratio $\gamma = m/n \in (0, 1]$. The empirical spectral distribution of the sample covariance matrix $\frac{1}{n} W W^T$ converges almost surely to the Marchenko-Pastur distribution $\mu_{\text{MP}}$ on the support $[\lambda_-, \lambda_+]$:
\begin{equation}
    \lambda_\pm = p \left(1 \pm \sqrt{\gamma}\right)^2
\end{equation}
and the extreme singular values satisfy:
\begin{equation}
    \frac{s_{\max}(W)}{\sqrt{n}} \xrightarrow{\text{a.s.}} \sqrt{p}\left(1 + \sqrt{\gamma}\right), \quad \frac{s_{\min}(W)}{\sqrt{n}} \xrightarrow{\text{a.s.}} \sqrt{p}\left(1 - \sqrt{\gamma}\right)
\end{equation}
\end{theorem}

\begin{remark}[Subspace Separability vs. Operator Norm Control]
An upper bound $s_{\max}(W) \le C$ prevents representation explosion, but does not prevent dimensional collapse (e.g., $s_{\min}(W) = 0$ annihilates orthogonal subspaces). To guarantee pairwise distance preservation, a trained BitLinear layer must satisfy restricted lower and upper singular bounds on the active semantic representation subspace $\mathcal{S}$:
\begin{equation}
    0 < c \cdot \|x - y\|_2 \le \|W(x - y)\|_2 \le C \cdot \|x - y\|_2, \quad \forall x, y \in \mathcal{S}
\end{equation}
We define the layer-wise spectral rigidity deviation $R_\ell$ relative to the Marchenko-Pastur null model:
\begin{equation}
    R_\ell = \max\left\{ \left|\frac{s_{\max}(W_\ell)^2}{n} - \lambda_+^{(\ell)}\right|,\, \left|\frac{s_{\min}(W_\ell)^2}{n} - \lambda_-^{(\ell)}\right| \right\}
\end{equation}
\end{remark}

\section{Structural Invariant Verification}
To evaluate behavioral correctness without relying solely on language model self-evaluation, we define an explicit four-predicate admissibility contract:
\begin{equation}
    \mathcal{A}(c) = \mathcal{P}(c) \land \mathcal{S}(c) \land \mathcal{T}(c) \land \mathcal{G}(c)
\end{equation}
where $\mathcal{P}(c)$ verifies concrete syntax tree parse validity without syntax errors; $\mathcal{S}(c)$ checks structural abstract syntax tree invariants (e.g., pure function signatures, absence of unauthorized state mutations); $\mathcal{T}(c)$ executes deterministic unit test assertions; and $\mathcal{G}(c)$ enforces isolated git worktree provenance and sandbox execution bounds.

\subsection{Throughput, Memory Footprint \& Hardware Specifications}
Benchmarking was executed on Apple M3 Max hardware ($128\text{ GB}$ Unified Memory, 16-core CPU / 40-core GPU, macOS 15.3, Metal Shading Language kernels):
\begin{itemize}
    \item \textbf{Resident Memory}: $3.4\text{ GB}$ (FP16 baseline) $\to$ $\mathbf{0.42\text{ GB}}$ ($8.1\times$ reduction, comprising $336\text{ MB}$ packed ternary codes, $28\text{ MB}$ group-64 scales, and $56\text{ MB}$ non-quantized RMSNorm/embeddings).
    \item \textbf{Inference Throughput}: $28\text{ tokens/s}$ (FP16 baseline) $\to$ $\mathbf{118\text{ tokens/s}}$ ($4.2\times$ speedup).
    \item \textbf{AST Verification}: $100.0\%$ pass rate observed across deterministic algorithmic benchmark suites ($0\%$ reward-hacking escapes under the verifier test suite).
\end{itemize}

\section{Conclusion}
\textsc{Kompress v2} unifies multi-teacher geometric-mean consensus distillation, random matrix spectral rigidity null models, and multi-agent AST worktree verification into a principled edge reasoning framework. By proving the exact equivalence between logit averaging and geometric-mean probability consensus, we provide a solid mathematical foundation for sovereign, ultra-low-latency edge intelligence.
